import {
  ShapeUtil, HTMLContainer, Rectangle2d, T, resizeBox,
  createShapePropsMigrationIds, createShapePropsMigrationSequence,
  type TLBaseShape, type RecordProps, type TLResizeInfo, type Geometry2d,
} from 'tldraw'
import katex from 'katex'
import 'katex/dist/katex.min.css'

export type MathShape = TLBaseShape<'math', { w: number; h: number; latex: string; fontSize: number }>

// version registry: bump when props change (see section 8)
export const mathVersions = createShapePropsMigrationIds('math', { Initial: 1 })

export class MathShapeUtil extends ShapeUtil<MathShape> {
  static override type = 'math' as const
  static override props: RecordProps<MathShape> = {
    w: T.nonZeroNumber, h: T.nonZeroNumber, latex: T.string, fontSize: T.positiveNumber,
  }
  static override migrations = createShapePropsMigrationSequence({
    sequence: [{ id: mathVersions.Initial, up: (props) => props, down: (props) => props }],
  })
  override getDefaultProps(): MathShape['props'] { return { w: 220, h: 64, latex: 'x^2 + 1', fontSize: 24 } }
  override getGeometry(shape: MathShape): Geometry2d {
    return new Rectangle2d({ width: shape.props.w, height: shape.props.h, isFilled: true })
  }
  override canResize() { return true }
  override isAspectRatioLocked() { return false }
  override canEdit() { return false }                 // set true + double-click editing later
  override onResize(shape: MathShape, info: TLResizeInfo<MathShape>) { return resizeBox(shape, info) }
  override component(shape: MathShape) {
    const html = katex.renderToString(shape.props.latex, { throwOnError: false, displayMode: true })
    return (
      <HTMLContainer style={{ display: 'flex', alignItems: 'center', justifyContent: 'center',
                               fontSize: shape.props.fontSize, overflow: 'hidden', pointerEvents: 'all' }}>
        <div dangerouslySetInnerHTML={{ __html: html }} />
      </HTMLContainer>
    )
  }
  override indicator(shape: MathShape) { return <rect width={shape.props.w} height={shape.props.h} rx={4} /> }
}

The draw tool: creates the shape at `x: originPagePoint.x, y: originPagePoint.y` with one segment whose first point is `{x:0,y:0,z:pressure}` (Drawing.mjs:170-193); each subsequent point is `editor.getPointInShapeSpace(shape, inputs.currentPagePoint).toFixed()` -> points are SHAPE-LOCAL (not page) and NOT scaled by `props.scale` (scale only affects stroke width: DrawShapeUtil.mjs:57-59; `scale = 1/zoom` only in dynamic-resize mode, Drawing.mjs:178); `z` is pressure (0.5 for mouse). Completion = one `updateShapes([{ id, type, props: { isComplete: true } }])` on pointer-up/complete (Drawing.mjs:503-527), all inside a history batch started by `markHistoryStoppingPoint('draw start')` (Drawing.mjs:108). Shift-click extends the SAME shape with another segment (Drawing.mjs:115-125), and `segments[].type === 'straight'` segments carry only 2 points (interpolate with `Vec.PointsBetween` like getPath.mjs:83-95 if your recognizer needs density). `TLDrawShapeProps = { color, fill, dash, size, segments: {type:'free'|'straight', points: VecModel[]}[], isComplete, isClosed, isPen, scale }` (tlschema d.ts:4055-4075, 4081-4084). Page transform: `editor.getShapePageTransform(shape): Mat` (editor d.ts:2517) and `Mat#applyToPoints(points): Vec[]` (4708).

```ts
import { useEffect } from 'react'
import { useEditor, type TLDrawShape, type TLRecord, type TLShape } from 'tldraw'

export type Stroke = { shapeId: string; points: { x: number; y: number; z: number }[]; bounds: Box }

function isDraw(r: TLRecord | undefined): r is TLDrawShape { return !!r && r.typeName === 'shape' && (r as TLShape).type === 'draw' }

export function useCompletedStrokes(onStrokes: (strokes: Stroke[]) => void) {
  const editor = useEditor()
  useEffect(() => {
    return editor.store.listen(({ changes }) => {
      const done: TLDrawShape[] = []
      for (const rec of Object.values(changes.added)) if (isDraw(rec) && rec.props.isComplete) done.push(rec)
      for (const [from, to] of Object.values(changes.updated))
        if (isDraw(to) && to.props.isComplete && !(isDraw(from) && from.props.isComplete)) done.push(to)
      if (!done.length) return
      onStrokes(done.map((shape) => {
        const t = editor.getShapePageTransform(shape.id)
        const points = shape.props.segments.flatMap((seg) => {
          const pagePts = t.applyToPoints(seg.points)             // page space (rotation/parent frames handled by the transform)
          return pagePts.map((p, i) => ({ x: p.x, y: p.y, z: seg.points[i].z ?? 0.5 }))
        })
        return { shapeId: shape.id, points, bounds: editor.getShapePageBounds(shape.id)! }
      }))
    }, { source: 'user', scope: 'document' })
  }, [editor, onStrokes])
}
```
For a streaming recognizer, also handle `updated` entries whose `to.props.isComplete === false` (partial strokes) and `changes.removed` (eraser). Alternative API: `editor.sideEffects.registerAfterChangeHandler('shape', (prev, next, source) => ...)` (store d.ts:2657; handler type `(prev, next, source: 'remote'|'user') => void`, 1798) fires synchronously per record inside the transaction — use it if frame-batched delivery is too coarse, but never mutate the store from it without care. Group strokes for math recognition by time gap + proximity of `bounds`, and send `[{x,y}[]]` to Mathpix `strokes` API (credentials already provisioned per CONTEXT.md).

5. SEPARATING AI-MADE CHANGES FROM STUDENT CHANGES
======================================================================
Every store mutation made outside `store.mergeRemoteChanges(fn)` has `source: 'user'` (Store.mjs:231, 265, 324), including your own `createShape`/`createAssets`. Options: (a) tag records (`meta: { origin: 'ai' }`) and filter the diff in listeners; (b) `editor.store.mergeRemoteChanges(() => { ...editor.createShape(...) })` marks them `'remote'` — invisible to `source:'user'` listeners and NOT recorded in undo (HistoryManager.mjs:34-35 `if (source !== "user") return;`). Do not rely on timers.

6. PAGE <-> SCREEN CONVERSION
======================================================================
Camera = `{x, y, z}` (`editor.getCamera()`, d.ts:1842; `getZoomLevel()` 1851). `getViewportPageBounds()` = `new Box(-cx, -cy, w/cz, h/cz)` (Editor.mjs:2829-2833); `getViewportScreenBounds()` = instance `screenBounds` (the tldraw container rect in window coords, 2821-2824). `pageToScreen(p)` = `((p.x + cx) * cz + screenBounds.x, (p.y + cy) * cz + screenBounds.y)` (2867-2875); `screenToPage(p)` = `((p.x - screenBounds.x) / cz - cx, ...)` (2846-2855); `pageToViewport(p)` = same as pageToScreen WITHOUT the screenBounds offset, i.e. relative to the tldraw container (2888-2891). `Box` exposes `x,y,w,h` plus `width/height/center/midX` getters (d.ts:458-483). For React overlays rendered inside `<Tldraw>` children (as BoardContent does) use `pageToViewport`, or render them in the `InFrontOfTheCanvas` component slot; use `pageToScreen` for `position: fixed` elements. In this app the wrapper is `position: fixed; inset: 0`, so screenBounds.x/y = 0 and both agree. Recompute on camera change with `useValue('cam', () => editor.getCamera(), [editor])` (re-exported from @tldraw/state-react). Bounds of a shape in page space: `editor.getShapePageBounds(id)` (2532).

7. HISTORY: markHistoryStoppingPoint / editor.run semantics
======================================================================
`markHistoryStoppingPoint(name?): string` (d.ts:1252; Editor.mjs:977-981) pushes a named mark onto the undo stack and clears redos; everything after it becomes the next undo step. Returned id works with `bailToMark(id)` (undo to the mark, discarding) and `squashToMark(id)` (collapse everything since the mark into one entry) (1261-1295). `run(fn, opts?: { history?: 'record' | 'record-preserveRedoStack' | 'ignore', ignoreShapeLock?: boolean })` (1320, 6686-6688, 7015-7022): runs `fn` in one history batch (one undo entry for all mutations), `history: 'ignore'` keeps the changes out of undo/redo entirely (still persisted and still delivered to store listeners), `record-preserveRedoStack` records without clearing redo, and `ignoreShapeLock: true` lets `updateShapes`/`deleteShapes` touch locked shapes and their descendants (Editor.mjs:1062-1070, 6109-6125, 6157-6165). Nested runs inherit the outer lock setting. `createAssets` is always `history: 'ignore'` (Editor.mjs:3375-3379). Draw tool pattern: mark at pointer-down, `bailToMark` on interrupt (Drawing.mjs:108, 513-518).

8. SNAPSHOT / MIGRATION IMPLICATIONS OF ADDING A SHAPE TYPE
======================================================================
Persisted snapshots carry `document.schema.sequences` (`{ 'com.tldraw.store': n, 'com.tldraw.shape.draw': n, ... }`). Every shape util (even with `migrations` undefined) gets a sequence `com.tldraw.shape.<type>` marked `retroactive: true` (recordsWithProps.mjs:6-16, 24-30). Loading an OLD board (no `com.tldraw.shape.math` key) into NEW code: `getMigrationsSince` includes the missing retroactive sequence and applies all of its migrations (StoreSchema.mjs:180-183, 193-197) — each is filtered to `r.typeName === 'shape' && r.type === 'math'` (recordsWithProps.mjs:63-66), so with zero math records it is a no-op: old boards open fine. Loading a NEW board (contains math shapes / `com.tldraw.shape.math: 1`) into OLD code without MathShapeUtil: unknown sequences are ignored by the migrator (StoreSchema.mjs:176-178) but `store.put` validation rejects the record (`T.union('type', ...)`: 'Expected one of "arrow" or ... , got "math"', validation.mjs:760) -> `loadSnapshot` throws (TLEditorSnapshot.mjs:9-13, Store.mjs:468-480) -> with the current onMount/catch the user gets an EMPTY board and autosave overwrites it (see high finding). Later prop changes: bump with `createShapePropsMigrationIds('math', { Initial: 1, AddFontSize: 2 })` and append `{ id: versions.AddFontSize, up: (props) => { props.fontSize ??= 24 }, down: (props) => { delete props.fontSize } }` to the sequence; a newer-versioned snapshot opened by older code fails with `Incompatible schema?` -> MigrationError (StoreSchema.mjs:199-204) — same failure mode, so: (1) ship the shape util to every <Tldraw> instance (board + /train) from one shared array, (2) gate autosave on successful load, (3) consider storing `schema` alongside a `client_version` and refusing to write when the stored schema has sequences the client lacks, (4) `getSnapshot(store)` returns `{ document, session }` (TLEditorSnapshot, d.ts:6691-6694) — session state (camera, selection, isDebugMode...) is currently persisted too; `loadSnapshot` accepts `Partial<TLEditorSnapshot> | TLStoreSnapshot` and merges session state unless `forceOverwriteSessionState` (7093-7100).
